def github = '''
import com.cloudbees.plugins.credentials.CredentialsProvider
import org.jenkinsci.plugins.plaincredentials.StringCredentials
import jenkins.model.Jenkins
import groovy.json.JsonSlurper

def token = CredentialsProvider.lookupCredentials(
	StringCredentials, Jenkins.instance, null, null
).find { it.id == 'github-token' }?.secret?.plainText

def get = { String path ->
	def conn = new URL('https://api.github.com' + path).openConnection()
	conn.setRequestProperty('Accept', 'application/vnd.github+json')
	if (token) conn.setRequestProperty('Authorization', 'Bearer ' + token)
    conn.connectTimeout = 5000
    conn.readTimeout = 20000
    conn.responseCode == 200 ? new JsonSlurper().parseText(conn.inputStream.text) : null
}
'''

pipelineJob('python-native-build') {
    parameters {
        activeChoiceParam('REPO') {
            choiceType('SINGLE_SELECT')
            groovyScript {
                script(github + '''
def repos = []
def page = 1
while (page <= 5) {
    def batch = get('/user/starred?per_page=100&page=' + page)
    if (!batch) break
    repos += batch.collect { it.full_name }
    if (batch.size() < 100) break
    page++
}
return repos ?: ['ERROR: could not list starred repositories']
''')
                fallbackScript('return ["ERROR: could not list repositories"]')
            }
        }
        activeChoiceReactiveParam('BRANCH') {
            choiceType('SINGLE_SELECT')
            referencedParameter('REPO')
            groovyScript {
                script(github + '''
if (!REPO || REPO.startsWith('ERROR')) return []
def head = get('/repos/' + REPO)?.default_branch
def names = get('/repos/' + REPO + '/branches?per_page=100')?.collect { it.name } ?: []
if (head && names.remove(head)) names.add(0, head)
return names ?: ['main']
''')
                fallbackScript('return ["main"]')
            }
        }
        activeChoiceReactiveParam('ENTRY') {
            choiceType('SINGLE_SELECT')
            referencedParameter('REPO')
            referencedParameter('BRANCH')
            groovyScript {
                script(github + '''
if (!REPO || !BRANCH || REPO.startsWith('ERROR')) return []
def tree = get('/repos/' + REPO + '/git/trees/' + BRANCH + '?recursive=1')
if (!tree) return ['ERROR: could not list files']

def files = tree.tree
        .findAll { it.type == 'blob' && it.path.endsWith('.py') }
        .collect { it.path }
return files
''')
                fallbackScript('return []')
            }
        }
      
      activeChoiceReactiveParam('DATA') {
            choiceType('MULTI_SELECT')
            referencedParameter('REPO')
            referencedParameter('BRANCH')
            groovyScript {
                script(github + '''
if (!REPO || !BRANCH || REPO.startsWith('ERROR')) return []
def tree = get('/repos/' + REPO + '/git/trees/' + BRANCH + '?recursive=1')
if (!tree) return ['ERROR: could not list files']

def skip = ['.git', '.github', '.venv', 'venv', '__pycache__', 'build', 'dist',
            'node_modules', '.tox', 'tests', 'test', 'docs']

return tree.tree
    .findAll { it.type == 'tree' }
    .collect { it.path }
    .findAll { d -> !d.tokenize('/').any { skip.contains(it) } }
    .sort()
''')
                fallbackScript('return []')
            }
        }
    }
  
        definition {
        cps {
            script('''
pipeline {
    agent { label 'rocky-node' }

    stages {
        stage('Build Image') {
            steps {
                sh 'rm -rf build src pipeline'
                sh 'git clone --depth 1 https://github.com/lberkk/nuitka-pipeline.git pipeline'
                sh 'docker build -t "nuitka-build:${BUILD_NUMBER}" pipeline'
            }
        }

        stage('Fetch') {
            steps {
                withCredentials([string(credentialsId: 'github-token', variable: 'GH_TOKEN')]) {
                    sh 'git clone --branch "$BRANCH" "https://x-access-token:$GH_TOKEN@github.com/$REPO.git" src'
                    sh 'git -C src remote set-url origin "https://github.com/$REPO.git"'
                }
            }
        }

        stage('Build') {
            steps {
                sh \'\'\'
                    ARGS=""
                    IFS=','
                    for d in $DATA; do
                        [ -n "$d" ] && ARGS="$ARGS --data-dir $d"
                    done
                    unset IFS

                    docker run --rm --user "$(id -u):$(id -g)" \\
                      -e HOME=/tmp \\
                      -v "$WORKSPACE:$WORKSPACE" -w "$WORKSPACE" "nuitka-build:${BUILD_NUMBER}" \\
                      python3.11 /opt/nuitka/nuitka_build.py "src/$ENTRY" --auto-deps $ARGS
                \'\'\'
            }
        }
    }

    post {
        success {
            archiveArtifacts artifacts: 'build/*'
        }
    }
}
''')
            sandbox(true)
        }
    }
}
